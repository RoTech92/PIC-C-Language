/* Program 8: Swap Two Numbers Using a Third Variable
   Turbo C++ version - already in Turbo C++ style, no changes needed */

#include<stdio.h>
#include<conio.h>

void main()
{
    int a, b, temp;

    clrscr();

    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);

    temp = a;
    a = b;
    b = temp;

    printf("After swap: a = %d, b = %d\n", a, b);

    getch();
}
