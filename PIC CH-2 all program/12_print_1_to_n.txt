/* Program 12: Print 1 2 3 4 ... N
   Turbo C++ version - added conio.h, clrscr(), getch(); void main() used */

#include<stdio.h>
#include<conio.h>

void main()
{
    int n, i;

    clrscr();

    printf("Enter value of N: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
        printf("%d ", i);

    getch();
}
