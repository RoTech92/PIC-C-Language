/* Program 14: Print 1 3 5 7 ... N (Odd Numbers)
   Turbo C++ version - added conio.h, clrscr(), getch(); void main() used */

#include<stdio.h>
#include<conio.h>

void main()
{
    int n, i;

    clrscr();

    printf("Enter value of N (count of odd numbers): ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
        printf("%d ", (2*i - 1));

    getch();
}
