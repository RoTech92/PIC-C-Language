/* Program 3: Check Even or Odd
   Turbo C++ version - added conio.h, clrscr(), getch(); void main() used */

#include<stdio.h>
#include<conio.h>

void main()
{
    int num;

    clrscr();

    printf("Enter a number: ");
    scanf("%d", &num);

    if(num % 2 == 0)
        printf("%d is Even", num);
    else
        printf("%d is Odd", num);

    getch();
}
