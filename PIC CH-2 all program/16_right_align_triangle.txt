/* Program 16: Right Align Triangle of Numbers
   Turbo C++ version - added conio.h, clrscr(), getch(); void main() used */

#include<stdio.h>
#include<conio.h>

void main()
{
    int n, i, j, k;

    clrscr();

    printf("Enter number of rows: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        for(j = 1; j <= (n - i); j++)
            printf(" ");            /* spaces for alignment */
        for(k = 1; k <= i; k++)
            printf("%d ", k);
        printf("\n");
    }

    getch();
}
