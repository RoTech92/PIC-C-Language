/* Program 27: Check Whether Number is Prime
   Turbo C++ version - added conio.h, clrscr(), getch(); void main() used */

#include<stdio.h>
#include<conio.h>

void main()
{
    int num, i, isPrime = 1;

    clrscr();

    printf("Enter a number: ");
    scanf("%d", &num);

    if(num <= 1)
        isPrime = 0;

    for(i = 2; i <= num/2; i++)
    {
        if(num % i == 0)
        {
            isPrime = 0;
            break;          /* no need to check further */
        }
    }

    if(isPrime == 1)
        printf("%d is Prime", num);
    else
        printf("%d is not Prime", num);

    getch();
}
