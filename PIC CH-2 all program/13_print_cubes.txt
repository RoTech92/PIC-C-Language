/* Program 13: Print 1 8 27 64 ... N (Cubes)
   Turbo C++ version - added conio.h, clrscr(), getch(); void main() used */

#include<stdio.h>
#include<conio.h>

void main()
{
    int n, i;

    clrscr();

    printf("Enter value of N: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
        printf("%d ", i*i*i);

    getch();
}
