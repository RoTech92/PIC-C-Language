/* Program 25: Check Whether Number is Palindrome
   Turbo C++ version - added conio.h, clrscr(), getch(); void main() used */

#include<stdio.h>
#include<conio.h>

void main()
{
    int num, original, digit, reverse = 0;

    clrscr();

    printf("Enter a number: ");
    scanf("%d", &num);

    original = num;
    while(num != 0)
    {
        digit = num % 10;
        reverse = reverse * 10 + digit;
        num = num / 10;
    }

    if(original == reverse)
        printf("%d is a Palindrome", original);
    else
        printf("%d is not a Palindrome", original);

    getch();
}
