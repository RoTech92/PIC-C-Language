/* Program 18: Print Multiplication Table of a Number
   Turbo C++ version - added conio.h, clrscr(), getch(); void main() used */

#include<stdio.h>
#include<conio.h>

void main()
{
    int num, i;

    clrscr();

    printf("Enter a number: ");
    scanf("%d", &num);

    for(i = 1; i <= 10; i++)
        printf("%d x %d = %d\n", num, i, num*i);

    getch();
}
